class _DashboardHeroPainter extends CustomPainter {
  const _DashboardHeroPainter({
    required this.panelColor,
    required this.glassColor,
    required this.lineColor,
    required this.barColor,
  });

  final Color panelColor;
  final Color glassColor;
  final Color lineColor;
  final Color barColor;

  @override
  void paint(Canvas canvas, Size size) {
    final panel = Paint()..color = panelColor;
    final glass = Paint()..color = glassColor;
    final line = Paint()
      ..color = lineColor
      ..strokeWidth = 2
      ..style = PaintingStyle.stroke;

    canvas.drawRRect(
      RRect.fromRectAndRadius(
        Rect.fromLTWH(size.width * .43, 18, size.width * .48, 106),
        const Radius.circular(8),
      ),
      panel,
    );
    canvas.drawRRect(
      RRect.fromRectAndRadius(
        Rect.fromLTWH(size.width * .48, 30, size.width * .38, 75),
        const Radius.circular(5),
      ),
      glass,
    );
    final chart = Path()
      ..moveTo(size.width * .52, 88)
      ..lineTo(size.width * .58, 70)
      ..lineTo(size.width * .63, 78)
      ..lineTo(size.width * .70, 52)
      ..lineTo(size.width * .77, 64)
      ..lineTo(size.width * .83, 42);
    canvas.drawPath(chart, line);
    for (var index = 0; index < 4; index++) {
      canvas.drawRect(
        Rect.fromLTWH(
          size.width * (.53 + index * .07),
          92 - index * 10,
          16,
          26 + index * 10,
        ),
        Paint()..color = barColor,
      );
    }
  }

  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => false;
}
